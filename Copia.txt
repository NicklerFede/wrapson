    function calcularTotalesVenta() {
      const lote = getLoteActivo();
      const cant = parseInt(document.getElementById("inputCantidad").value) || 1;
      const pv = lote.precioVenta;

      document.getElementById("badgePrecioVenta").textContent = `$${pv.toLocaleString('es-AR')} c/u`;
      document.getElementById("labelDetalleCobro").textContent = `${cant} wrap${cant > 1 ? 's' : ''} × $${pv.toLocaleString('es-AR')}`;
      document.getElementById("labelTotalCobrar").textContent = `$${(cant * pv).toLocaleString('es-AR')}`;
      calcularVuelto();
    }

    function calcularVuelto() {
      const lote = getLoteActivo();
      const cant = parseInt(document.getElementById("inputCantidad").value) || 1;
      const pv = lote.precioVenta;
      const pago = parseFloat(document.getElementById("inputPagoCliente").value) || 0;
      document.getElementById("labelMontoVuelto").textContent = `$${Math.max(0, pago - (cant * pv)).toLocaleString('es-AR')}`;
    }

    function guardarVenta() {
      const lote = getLoteActivo();
      const cliente = document.getElementById("inputCliente").value.trim() || "Cliente Anónimo";
      const fecha = document.getElementById("inputFecha").value.trim() || obtenerFechaHoy();
      const cantidad = parseInt(document.getElementById("inputCantidad").value) || 1;
      const pagoCliente = parseFloat(document.getElementById("inputPagoCliente").value) || (cantidad * lote.precioVenta);

      if (!window.estadoApp.ventas) window.estadoApp.ventas = [];
      window.estadoApp.ventas.unshift({
        id: Date.now(),
        loteId: lote.id,
        cliente,
        variedad: variedadSeleccionada,
        cantidad,
        precioUnitario: lote.precioVenta,
        precioCompra: lote.precioCompra,
        metodoPago: metodoPagoSeleccionado,
        montoVuelto: Math.max(0, pagoCliente - (cantidad * lote.precioVenta)),
        estadoPago: document.getElementById("selectEstadoPago").value,
        estadoEntrega: document.getElementById("selectEstadoEntrega").value,
        fecha
      });

      guardarEnNube();
      document.getElementById("inputCliente").value = "";
      document.getElementById("inputPagoCliente").value = "";
      setCantidad(1);
      renderizarTodo();
    }

    function renderizarTablaStock() {
      const tbody = document.getElementById("tablaStock");
      if (!tbody) return;
      tbody.innerHTML = "";
      const lote = getLoteActivo();
      const ventas = window.estadoApp.ventas || [];
      VARIEDADES.forEach(v => {
        const inicial = (lote.stockInicial && lote.stockInicial[v]) || 0;
        const vendidos = ventas.filter(x => x.loteId === lote.id && x.variedad === v).reduce((a, c) => a + c.cantidad, 0);
        const disp = inicial - vendidos;
        tbody.innerHTML += `<tr class="hover:bg-slate-800/50"><td class="py-2.5 px-3 font-semibold text-slate-200">${v}</td><td class="py-2.5 px-3 text-center text-slate-400">${inicial}</td><td class="py-2.5 px-3 text-center text-slate-400">${vendidos}</td><td class="py-2.5 px-3 text-center font-bold ${disp <= 0 ? 'text-rose-400' : 'text-emerald-400'}">${disp}</td></tr>`;
      });
    }

    function renderizarRendimientoSabores() {
      const container = document.getElementById("contenedorRendimientoSabores");
      if (!container) return;
      container.innerHTML = "";

      const lote = getLoteActivo();
      const ventas = (window.estadoApp.ventas || []).filter(v => v.loteId === lote.id);
      document.getElementById("textoLoteActivoRendimiento").textContent = lote.id;

      const datosSabores = VARIEDADES.map(v => {
        const inicial = (lote.stockInicial && lote.stockInicial[v]) || 0;
        const vendidos = ventas.filter(x => x.variedad === v).reduce((a, c) => a + c.cantidad, 0);
        const porcentaje = inicial > 0 ? Math.min(100, Math.round((vendidos / inicial) * 100)) : 0;
        return { variedad: v, inicial, vendidos, porcentaje };
      }).sort((a, b) => b.porcentaje - a.porcentaje);

      datosSabores.forEach(item => {
        container.innerHTML += `
          <div class="space-y-1">
            <div class="flex justify-between items-center text-xs">
              <span class="font-bold text-slate-200">${item.variedad}</span>
              <span class="font-black text-amber-400">${item.porcentaje}% <span class="text-slate-400 font-normal">(${item.vendidos}/${item.inicial} un.)</span></span>
            </div>
            <div class="w-full bg-slate-950 rounded-full h-2.5 p-0.5 border border-slate-800 overflow-hidden">
              <div class="bg-gradient-to-r from-amber-600 to-amber-400 h-full rounded-full transition-all duration-500" style="width: ${item.porcentaje}%"></div>
            </div>
          </div>
        `;
      });
    }

    function renderizarResumenFinanciero() {
      const lote = getLoteActivo();
      const ventasLote = (window.estadoApp.ventas || []).filter(v => v.loteId === lote.id);
      let totalVendidos = 0, gananciaBruta = 0, costoCompra = 0, efectivoIngresado = 0, transfIngresadas = 0, vueltos = 0, fiado = 0;

      ventasLote.forEach(v => {
        totalVendidos += v.cantidad;
        const sub = v.cantidad * (v.precioUnitario || lote.precioVenta);
        const cos = v.cantidad * (v.precioCompra || lote.precioCompra);
        if (v.estadoPago === 'Pagado') {
          gananciaBruta += sub;
          costoCompra += cos;
          if (v.metodoPago === 'Efectivo') efectivoIngresado += sub; else transfIngresadas += sub;
          vueltos += (v.montoVuelto || 0);
        } else {
          fiado += sub;
        }
      });

      document.getElementById("resumenVendidos").textContent = `${totalVendidos} un.`;
      document.getElementById("resumenBruta").textContent = `$${gananciaBruta.toLocaleString('es-AR')}`;
      document.getElementById("resumenNeta").textContent = `$${(gananciaBruta - costoCompra).toLocaleString('es-AR')}`;
      document.getElementById("resumenCajaReal").textContent = `$${(efectivoIngresado - vueltos).toLocaleString('es-AR')}`;
      document.getElementById("resumenTransfNeta").textContent = `$${transfIngresadas.toLocaleString('es-AR')}`;
      document.getElementById("resumenFiado").textContent = `$${fiado.toLocaleString('es-AR')}`;
    }

    function importarJSON(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = async function (e) {
        try {
          const data = JSON.parse(e.target.result);
          if (!window.estadoApp.lotes) window.estadoApp.lotes = [];
          if (!window.estadoApp.ventas) window.estadoApp.ventas = [];

          let precioVentaDefecto = 5700;
          let precioCompraDefecto = 2650;
          if (data.precios) {
            if (data.precios.venta) precioVentaDefecto = data.precios.venta;
            if (data.precios.compra) precioCompraDefecto = data.precios.compra;
          }

          if (data.lotes && Array.isArray(data.lotes)) {
            data.lotes.forEach(loteNuevo => {
              if (loteNuevo.precioVenta === undefined) loteNuevo.precioVenta = precioVentaDefecto;
              if (loteNuevo.precioCompra === undefined) loteNuevo.precioCompra = precioCompraDefecto;

              const existe = window.estadoApp.lotes.some(l => l.id === loteNuevo.id);
              if (!existe) {
                window.estadoApp.lotes.push(loteNuevo);
              }
            });
          }

          if (data.ventas && Array.isArray(data.ventas)) {
            data.ventas.forEach(ventaNueva => {
              if (ventaNueva.precioUnitario === undefined) ventaNueva.precioUnitario = precioVentaDefecto;
              if (ventaNueva.precioCompra === undefined) ventaNueva.precioCompra = precioCompraDefecto;
              if (!ventaNueva.id) ventaNueva.id = Date.now() + Math.random();

              const existeVenta = window.estadoApp.ventas.some(v => v.id === ventaNueva.id);
              if (!existeVenta) {
                window.estadoApp.ventas.push(ventaNueva);
              }
            });
          }

          if (data.imagen1) window.estadoApp.imagen1 = data.imagen1;
          if (data.imagen2) window.estadoApp.imagen2 = data.imagen2;

          await guardarEnNube();
          renderizarSelectorLotes();
          renderizarTodo();
          alert("¡Lotes, ventas e información importados con éxito!");
        } catch (err) {
          console.error(err);
          alert("Error al leer o procesar el archivo JSON.");
        }
      };
      reader.readAsText(file);
    }
